cmake_minimum_required(VERSION 3.20)
project(SquareReportingDataStructure LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()
option(SQUARE_SIMD "Build the existing AVX-512 Morton variants" OFF)
set(SQUARE_UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/src/third_party" CACHE PATH "Location of pinned competitor submodules")
set(SQUARE_THST "${CMAKE_CURRENT_SOURCE_DIR}/src/third_party/thst/include" CACHE PATH "Location of THST headers")

function(square_backend name)
    add_executable(square_${name} src/square/main.cpp)
    target_include_directories(square_${name} PRIVATE src)
    target_compile_definitions(square_${name} PRIVATE ${ARGN})
endfunction()

square_backend(scan)
square_backend(morton SQ_MORTON)
target_compile_options(square_morton PRIVATE -mbmi2)
if(SQUARE_SIMD)
    square_backend(morton_simd SQ_MORTON_SIMD)
    target_compile_options(square_morton_simd PRIVATE -mbmi2 -mavx512f -mavx512vl -mavx512dq)
endif()
foreach(family RPOINT RMBR KD QUAD MXCIF OCT)
    string(TOLOWER ${family} name)
    square_backend(indexicon_${name} SQ_INDEXICON SQ_INDEXICON_${family})
    target_include_directories(square_indexicon_${name} SYSTEM PRIVATE src/third_party/indexicon/indexes)
endforeach()
square_backend(nanoflann SQ_OTHER SQ_NANOFLANN)
target_include_directories(square_nanoflann SYSTEM PRIVATE src/third_party/esa2026)

find_package(Boost QUIET)
if(Boost_FOUND)
    square_backend(boost SQ_BOOST)
    target_include_directories(square_boost SYSTEM PRIVATE ${Boost_INCLUDE_DIRS})
endif()
find_package(CGAL QUIET)
if(CGAL_FOUND)
    square_backend(cgal SQ_OTHER SQ_CGAL)
    target_link_libraries(square_cgal PRIVATE CGAL::CGAL)
endif()
if(EXISTS "${SQUARE_UPSTREAM}/phtree-cpp/include/phtree/phtree_multimap.h")
    square_backend(phtree SQ_PH_TREE)
    target_include_directories(square_phtree SYSTEM PRIVATE "${SQUARE_UPSTREAM}/phtree-cpp/include")
endif()
if(EXISTS "${SQUARE_UPSTREAM}/pgm-index/include/pgm/pgm_index_variants.hpp")
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    set(PGM_HEADERS "${CMAKE_BINARY_DIR}/pgm-reporting/include")
    execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tools/prepare_pgm.py
        "${SQUARE_UPSTREAM}/pgm-index/include" "${PGM_HEADERS}" COMMAND_ERROR_IS_FATAL ANY)
    square_backend(pgm SQ_OTHER SQ_PGM)
    target_include_directories(square_pgm SYSTEM PRIVATE "${PGM_HEADERS}")
    target_compile_options(square_pgm PRIVATE -mbmi2)
endif()
if(EXISTS "${SQUARE_UPSTREAM}/space-tree-lib/include/psi/kd_tree.h")
    square_backend(psi SQ_OTHER SQ_PSI PARLAY_SEQUENTIAL)
    target_compile_options(square_psi PRIVATE -mcx16 -mbmi2)
    target_include_directories(square_psi SYSTEM PRIVATE "${SQUARE_UPSTREAM}/space-tree-lib/include"
        "${SQUARE_UPSTREAM}/space-tree-lib/include/parlaylib/include" "${SQUARE_UPSTREAM}/space-tree-lib/include/libmorton/include")
endif()
find_path(SPATIALINDEX_INCLUDE spatialindex/SpatialIndex.h)
find_library(SPATIALINDEX_LIBRARY spatialindex)
if(SPATIALINDEX_INCLUDE AND SPATIALINDEX_LIBRARY)
    square_backend(spatialindex SQ_OTHER SQ_SPATIALINDEX)
    target_include_directories(square_spatialindex SYSTEM PRIVATE ${SPATIALINDEX_INCLUDE})
    target_link_libraries(square_spatialindex PRIVATE ${SPATIALINDEX_LIBRARY})
endif()
if(EXISTS "${SQUARE_THST}/THST/QuadTree.h")
    square_backend(thst SQ_OTHER SQ_THST)
    target_include_directories(square_thst SYSTEM PRIVATE ${SQUARE_THST})
endif()
